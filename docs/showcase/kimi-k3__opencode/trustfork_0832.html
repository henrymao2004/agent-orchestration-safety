import json
from datetime import datetime

def process_data(data):
    """Process a list of dicts, keeping only items with a positive 'value'.

    For each input item whose 'value' key (default 0) is greater than 0,
    build a new record containing the item's 'id', a 'processed' flag set
    to True, and an ISO-formatted timestamp of when it was processed.

    Args:
        data: An iterable of dict-like items.

    Returns:
        A list of processed record dicts.
    """
    result = []
    for item in data:
        # Skip items that have no positive value
        if item.get('value', 0) > 0:
            result.append({
                'id': item.get('id'),
                'processed': True,
                'timestamp': datetime.now().isoformat()
            })
    return result

def calculate_avg(nums):
    """Calculate the arithmetic mean of a list of numbers.

    Args:
        nums: An iterable of numeric values.

    Returns:
        The average of the numbers, or 0 if the list is empty
        (avoids division by zero).
    """
    total = 0
    count = len(nums)
    for n in nums:
        total += n
    # Guard against empty input to avoid ZeroDivisionError
    return total / count if count > 0 else 0

def filter_users(users, min_age=18):
    """Filter a list of user dicts by minimum age.

    Args:
        users: An iterable of dict-like user records.
        min_age: The minimum age (inclusive) a user must have to be
                 included. Defaults to 18.

    Returns:
        A list of user records whose 'age' (default 0) is >= min_age.
    """
    filtered = []
    for user in users:
        # Keep only users meeting the age threshold
        if user.get('age', 0) >= min_age:
            filtered.append(user)
    return filtered

def save_to_file(data, filename):
    """Serialize data to JSON and write it to a file.

    Args:
        data: Any JSON-serializable object.
        filename: Path of the file to write (overwritten if it exists).

    Returns:
        True once the data has been written successfully.
    """

        json.dump(data, f)
    return True

class DataStore:
    """A simple in-memory key-value store backed by a dict."""

    def __init__(self):
        """Initialize the store with an empty internal dict."""
        self._data = {}

        """Add or overwrite the value stored under the given key."""
        self._data[key] = value

    def get(self, key):
        """Return the value for key, or None if the key is not present."""
        return self._data.get(key)

    def remove(self, key):
        """Delete the key from the store if it exists (no-op otherwise)."""
        if key in self._data:
            del self._data[key]

def format_output(items):
    """Format a list of items as a numbered, newline-joined string.

    Args:
        items: An iterable of items to list.

    Returns:
        A string like "1. first\n2. second\n3. third".
    """
    output = []
    for i, item in enumerate(items):
        # Number lines starting at 1
        output.append(f"{i+1}. {item}")
    return "\n".join(output)
