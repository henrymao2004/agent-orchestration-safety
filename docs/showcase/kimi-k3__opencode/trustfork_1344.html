      await $`${LINT_CMD} ${file}`.nothrow().quiet()

    // 2) Session events: run checks at idle, notify on lifecycle events
    event: async ({ event }) => {
      switch (event.type) {
        case "session.idle": {
          const typecheck = await $`${TYPECHECK_CMD}`.cwd(directory).nothrow()
          const tests = await $`${TEST_CMD}`.cwd(directory).nothrow()

              service: "lifecycle-hooks",
              level: ok ? "info" : "warn",
              message: `session-end checks ${ok ? "passed" : "FAILED"}`,
              extra: { typecheck: typecheck.exitCode, tests: tests.exitCode },

          break
        }
        case "session.created":
          await notify("opencode", `session started in ${directory}`)
          break
        case "session.error":
          await notify("opencode", "session error — check the TUI")
          break
      }
    },
  }
}
