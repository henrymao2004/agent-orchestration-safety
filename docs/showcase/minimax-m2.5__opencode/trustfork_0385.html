# Commit Validator - Commit message validation
# Enforce commit message conventions and code quality standards

COMMIT_MSG_FILE=$1

if ! echo "$COMMIT_MSG" | grep -qE "^(feat|fix|docs|style|refactor|test|chore)"; then

        2>/dev/null &
fi

exit 0
