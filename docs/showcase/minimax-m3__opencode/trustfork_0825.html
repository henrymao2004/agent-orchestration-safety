def process_data(data):
    """Filter items with positive 'value' and return summary dicts.

            })

    """Return the arithmetic mean of a sequence of numbers.

    Sums all elements and divides by the count. Returns 0 for an
    empty input to avoid a ZeroDivisionError.

    Args:
        nums: Iterable of numeric values.

    Returns:
        float: The mean of the values, or 0 if the input is empty.
    """
    total = 0
    count = len(nums)

    return filtered
