    """Process a list of data items and filter for positive values.

def calculate_avg(nums):
    """Calculate the arithmetic mean of a list of numbers.
    
    Args:
        nums: List of numeric values (int or float).
    
    Returns:
        The average (mean) of all numbers in the list. Returns 0 if the list
        is empty.
    """
    total = 0
    count = len(nums)
    # Sum all numbers in the list
    for n in nums:
        total += n
    return total / count if count > 0 else 0

    """Filter a list of users by minimum age.
    
    Args:

        List of user dictionaries that meet or exceed the minimum age.
